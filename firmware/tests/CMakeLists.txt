cmake_minimum_required(VERSION 3.20)
project(esc_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(test_logic test_logic.c ../src/commands.c ../src/measurements.c)
target_include_directories(test_logic PRIVATE ../include)
add_test(NAME commands_and_measurements COMMAND test_logic)
add_executable(test_safe_io test_safe_io.c ../src/safe_io.c)
target_include_directories(test_safe_io PRIVATE mock ../include)
add_test(NAME reset_output_defaults COMMAND test_safe_io)
add_executable(test_console test_console.c ../src/console.c ../src/commands.c)
target_include_directories(test_console PRIVATE mock ../include)
add_test(NAME console_boundaries COMMAND test_console)
foreach(t test_logic test_safe_io test_console)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(${t} PRIVATE -fsanitize=address,undefined)
endforeach()
